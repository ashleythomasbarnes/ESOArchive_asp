-- Use case name: 'cat tap_schema 1a'
-- Which tables can be queried via this service? Show me only the most recent versions. 

SELECT schema_name, collection, table_name, title, version -- you could add: ,description, publication_date, number_rows, rel_descr_url, etc.
FROM TAP_SCHEMA.tables
WHERE cat_id in 
             (select t1.cat_id cat_id 
              from TAP_SCHEMA.tables t1
              left outer join TAP_SCHEMA.tables t2 on (t1.title = t2.title and t1.version< t2.version) 
              where t2.title is null)
ORDER BY schema_name, collection, title

-- Note: this query returns the most recent version of any catalog.
-- Previous versions of the catalogs might also be available, see next query example (cts1b).
-- Full explanation:
-- All TAP services must support a set of (meta) tables, in a schema named TAP_SCHEMA.
-- These meta-tables describe all tables and columns published through the TAP service.
-- The meta-tables can be queried themselves via TAP.
-- Users can discover ESO-published tables or columns by querying the various tables in the TAP_SCHEMA.

-- Use case name: 'cat tap_schema 2'
-- I am interested in the COMOS2015 data. What are the column names of the COSMOS2015 table? 

SELECT column_name, datatype, unit, ucd -- add description to know more
from TAP_SCHEMA.columns
where table_name='COSMOS2015_Laigle_v1_1b_latestV7_fits_V1'

-- WHAT ARE THE COLUMNS OF THE DATA PRODUCT TABLE called COSMOS2015_Laigle_v1_1b_latestV7_fits_V1?
-- The TAP_SCHEMA.columns table contains all column names, units,
-- UCDs, description, etc., of all the published ESO tables.

-- Use case name: 'cat tap_schema 3'
-- What are the names of the columns providing the RA and DEC in the PESSTO catalog?

SELECT column_name
from TAP_SCHEMA.columns
where table_name='PESSTO_TRAN_CAT_fits_V2'
and (ucd = 'pos.eq.ra;meta.main' or ucd = 'pos.eq.dec;meta.main')

-- The ucd tells which columns are carrying the main coordinates.

-- Use case name: 'cat tap_schema 4'
-- What are the names of the columns providing: target names, RA, DEC for all the VMC catalogs? 

SELECT publication_date, T.table_name, column_name, ucd
FROM TAP_SCHEMA.columns as C, TAP_SCHEMA.tables as T
where T.table_name=C.table_name and cat_id in (SELECT cat_id from TAP_SCHEMA.tables where collection='VMC')
and (ucd = 'pos.eq.ra;meta.main' or ucd = 'pos.eq.dec;meta.main' or ucd = 'meta.id;meta.main')
order by publication_date desc, 2, 4

-- The ucd tells which columns are carrying the main coordinates and source identifier.


-- Use case name: 'cat tap_schema 5'
-- What are the names of the columns providing target name, RA, DEC for all the catalogs? 

SELECT publication_date, collection, T.table_name, column_name, ucd
FROM TAP_SCHEMA.columns as C, TAP_SCHEMA.tables as T
where T.table_name=C.table_name
and (ucd = 'pos.eq.ra;meta.main' or ucd = 'pos.eq.dec;meta.main' or ucd = 'meta.id;meta.main')
order by publication_date desc, 2,3,5

-- The ucd tells which columns are carrying the main coordinates and source identifier.

-- Use case name: 'cat metajoin 1'
-- I want to know which tables can be joined with the table VVV_CAT_V2

SELECT k.from_table, kc.from_column, k.target_table, kc.target_column
from TAP_SCHEMA.columns as c, TAP_SCHEMA.keys as k, TAP_SCHEMA.key_columns as kc
where  c.table_name='VVV_CAT_V2'
AND (
       (k.from_table=c.table_name and kc.from_column=c.column_name)
       OR
       (k.target_table=c.table_name and kc.target_column=c.column_name)
    )
AND k.key_id=kc.key_id
order by table_name, column_name, 1, 3, 2

-- Search for VVV tables that can be joined with VVV_CAT_V2
-- The results show that the joining condition is between:

-- Use case name: 'cat spatial_query 1'
-- Serach by position: Search the pessto master catalog around a given position

SELECT host_id, transient_id, transient_classification, transient_raj2000, transient_decj2000
FROM pessto_tran_cat_fits_V2
WHERE CONTAINS(point('', transient_raj2000, transient_decj2000), circle('', 41.2863, -55.7406, 0.04))=1

-- Search the PESSTO master catalog for transients in a circle of 0.04 deg (2.4 arcmin)
-- aroung galaxy 'ESO 154-10', which ICRS position is (41.2863, -55.7406)


-- Use case name: 'cat lightcurve 1'
-- HOw to get a light curve of a VVV object? 

SELECT top 10 VVV.IAUNAME, VVV.SOURCEID, VVV.RA2000, VVV.DEC2000,
              M.MJD, M.KSMAG, M.KSERR, 
              V.KSMEANMAG, V.KSAMPL, KSPROBVAR
FROM VVV_CAT_V2 as VVV
JOIN VVV_MPHOT_Ks_V2 as M  ON VVV.SOURCEID = M.SOURCEID
JOIN VVV_VAR_V2 as V       ON VVV.SOURCEID = V.SOURCEID
WHERE VVV.IAUNAME='VVV J135433.73-594836.43'
ORDER BY 1, 5

-- Alias to VVV: the VVV master catalog,
-- Alias to M: the table that contains the multi-epoch measurements of the object
-- Alias to V: the table that contains the variability information of the object


-- Use case name: 'cat lightcurve 2'
-- How to get the light curve of PESSTO transients at a certain sky position? 

SELECT host_id, transient_id, transient_classification,
       lc.B_VEGA_MAG, lc.B_VEGA_MAGERR,
       lc.R_AB_MAG, lc.R_AB_MAGERR, 
       lc.R_VEGA_MAG, lc.R_VEGA_MAGERR
FROM safcat.PESSTO_TRAN_CAT_V3 as master
JOIN safcat.PESSTO_MPHOT_V3 as lc  ON lc.SOURCE_ID = master.TRANSIENT_ID
WHERE
CONTAINS(point('', transient_raj2000, transient_decj2000), circle('', 41.2863, -55.7406, 0.05))=1

-- Search some magnitudes from the PESSTO catalogue of lightcurves (PESSTO_MPHOT_V3) 
-- for transients in a circle of 0.05 deg (3 arcmin) aroung galaxy 'ESO 154-10', positioned at (41.2863, -55.7406).
-- The lightcurve catalogue does not provide coordinates, hence the join with the master PESSTO catalogue.
-- To get all existing magnitudes (not only B and R), use lc.* in the SELECT part.
